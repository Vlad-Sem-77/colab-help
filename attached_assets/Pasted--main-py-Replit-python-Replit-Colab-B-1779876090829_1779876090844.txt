def check_colab_health(url):
    """Проверка соединения с Colab"""
    try:
        response = requests.get(f"{url}/health", timeout=5)
        return response.status_code == 200
    except:
        return False

def load_free_models(url):
    """Загрузка списка бесплатных моделей через Colab"""
    try:
        response = requests.get(f"{url}/models/free", timeout=10)
        if response.status_code == 200:
            data = response.json()
            return data.get("models", [])
        return []
    except Exception as e:
        st.error(f"Ошибка загрузки моделей: {e}")
        return []

def send_to_colab(url, payload):
    """Отправка задания в Colab"""
    try:
        response = requests.post(f"{url}/run", json=payload, timeout=60)
        if response.status_code == 200:
            return response.json()
        else:
            return {"status": "error", "message": f"HTTP {response.status_code}: {response.text}"}
    except requests.exceptions.Timeout:
        return {"status": "error", "message": "Превышено время ожидания (60 сек)"}
    except Exception as e:
        return {"status": "error", "message": str(e)}

print("\\nСравнение завершено.")

'''
    else:

def generate_txt_report(task_type, mode, model_card, params, metrics, result_text):
    """Генерация TXT-отчёта"""
    report = f"""============================================================
Replit ↔ Colab Bridge — Отчёт об эксперименте
============================================================
Дата: {datetime.now().strftime("%Y-%m-%d %H:%M:%S")}
Задание: {task_type}
Режим: {"Автовыбор free-модели" if mode == "auto" else "Ручной выбор"}
"""

    if model_card:
        report += f"""
--- Карточка модели ---
Название: {model_card.get('name', 'N/A')}
Провайдер: {model_card.get('top_provider', 'N/A')}
Контекст: {model_card.get('context_length', 'N/A')} токенов
Цена: $0.00 / млн токенов
Рейтинг: {model_card.get('average_rating', 'N/A')} / 5.0
"""

    report += f"""
--- Параметры ---
"""
    for key, value in params.items():
        report += f"{key}: {value}\n"

    if metrics:
        report += f"""
--- Метрики ---
Токенов запроса: {metrics.get('prompt_tokens', 'N/A')}
Токенов ответа: {metrics.get('completion_tokens', 'N/A')}
Всего токенов: {metrics.get('total_tokens', 'N/A')}
Время выполнения: {metrics.get('execution_time', 'N/A')} сек
"""

    report += f"""
--- Ответ модели ---
{result_text}
============================================================
"""
    return report
